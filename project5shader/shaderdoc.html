void mainImage( out vec4 fragColor, in vec2 fragCoord )
{
    vec2 uv_origin = (fragCoord - 0.5 * iResolution.xy) / iResolution.y;
    vec3 finalColor = vec3(0.0);

    for(float i = 0.0; i < 3.0; i++) {
        
        vec2 uv = uv_origin * (1.0 + i * 0.1); 

        float r = length(uv);
        float a = atan(uv.y, uv.x);
        float sides = 6.0;
        float tau = 6.283185;
        a = mod(a, tau / sides);
        a = abs(a - tau / sides / 2.0);
        uv = vec2(cos(a), sin(a)) * r;

        float t = iTime * (0.5 + i * 0.2);
        vec2 steppedUV = floor(uv * 15.0) / 15.0;
        vec2 gv = mod(uv * 15.0, 1.0) - 0.5;
        
        float dist = length(gv);
        float angle = atan(gv.y, gv.x);
        float seed = hash(steppedUV + i);
        
        float shapeNoise = sin(angle * 5.0 + seed * 6.28 + t) * 0.1;
        float baseRadius = 0.1 + seed * 0.3 + sin(t + seed * 10.0) * 0.05;
        
        float mask = step(dist + shapeNoise, baseRadius);
        
        vec3 layerColor = mix(vec3(1.0, 0.5, 0.2), vec3(0.2, 0.6, 1.0), length(uv) + i*0.2);
        
        finalColor += layerColor * mask * (1.0 - i * 0.3);
    }

    fragColor = vec4(finalColor, 1.0);
}
